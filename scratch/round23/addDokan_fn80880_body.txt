void dBg_ctr_c::addDokanMoveDiff(mVec3_c *out) {
    if (m_c4 == 0) return;

    f32 dx = out->x - mPos.x;
    f32 dy = out->y - mPos.y;

    f32 len = EGG::Math<f>::sqrt(dx * dx + dy * dy);

    s16 angle = static_cast<s16>(nw4r::math::Atan2Idx(dy, dx) + m_c4);

    f32 cos = nw4r::math::CosFIdx(static_cast<f32>(angle) * (1.0f / 65536.0f));
    f32 sin = nw4r::math::SinFIdx(static_cast<f32>(angle) * (1.0f / 65536.0f));

    // corrected delta (dead stores — present in the target as-is)
    f32 corrected_dx = len * cos - dx;
    f32 corrected_dy = len * sin - dy;

    out->x = dScStage_c::getLoopPosX(mPos.x + len * nw4r::math::CosFIdx(static_cast<f32>(angle) * (1.0f / 65536.0f)));
    out->y = mPos.y + len * nw4r::math::SinFIdx(static_cast<f32>(angle) * (1.0f / 65536.0f));
}
